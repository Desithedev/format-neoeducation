\newlength{\NEOCodeWidth}
\newcommand{\NEOCodeCell}[1]{
  \begin{minipage}[t]{\linewidth}
    \vspace{0pt}
    \ifdim\NEOCodeWidth>\linewidth
      \resizebox{\linewidth}{!}{\makebox[\NEOCodeWidth][l]{\usebox{#1}}}
    \else
      \usebox{#1}
    \fi
    \par\vspace{4pt}
  \end{minipage}
}
\newcommand{\NEOCode}[1]{
  \begingroup
  \sbox{\NEOCodeBox}{\BVerbatimInput[fontsize=\small,formatcom=\color{black},tabsize=4,showspaces=false,showtabs=false]{#1}}
  \setlength{\NEOCodeWidth}{\wd\NEOCodeBox}
  \NEOCodeCell{\NEOCodeBox}
  \endgroup
}
\newcommand{\NEOCodePair}[2]{
  \par\vspace{0.1cm}
  \begingroup
  \sbox{\NEOCodeBox}{\BVerbatimInput[fontsize=\small,formatcom=\color{black},tabsize=4,showspaces=false,showtabs=false]{#1}}
  \sbox{\NEOCodeRightBox}{\BVerbatimInput[fontsize=\small,formatcom=\color{black},tabsize=4,showspaces=false,showtabs=false]{#2}}
  \setlength{\NEOCodeWidth}{\wd\NEOCodeBox}
  \ifdim\wd\NEOCodeRightBox>\NEOCodeWidth
    \setlength{\NEOCodeWidth}{\wd\NEOCodeRightBox}
  \fi
  \renewcommand{\arraystretch}{1.2}
  \noindent\begin{tabularx}{\linewidth}{|>{\raggedright\arraybackslash}X|>{\raggedright\arraybackslash}X|}
    \hline
    \centering\textbf{Đoạn chương trình viết bằng Python} & \centering\arraybackslash\textbf{Đoạn chương trình viết bằng C++} \\
    \hline
    \NEOCodeCell{\NEOCodeBox} & \NEOCodeCell{\NEOCodeRightBox} \\
    \hline
  \end{tabularx}
  \endgroup
  \par\vspace{0.1cm}
}
